-- [[ ZALGO GENERATOR EFEKTU ]]
local function toZalgo(text)
	local zalgoChars = {"\204\141", "\204\142", "\204\143", "\204\144", "\204\145", "\204\146", "\204\147", "\204\148", "\204\149", "\204\150", "\204\151", "\204\152", 
		"\204\153", "\204\154", "\204\155", "\204\156", "\204\157", "\204\158", "\204\159", "\204\160"}
	local result = ""
	for i = 1, #text do
		local char = text:sub(i, i)
		result = result .. char
		for _ = 1, math.random(1, 4) do
			result = result .. zalgoChars[math.random(1, #zalgoChars)]
		end
	end
	return result
end

-- [[ NUMBER FORMATTER ]]
local function FormatAbbreviated(number)
	if number >= 1000000000 then
		return string.format("%.2fB", number / 1000000000)
	elseif number >= 1000000 then
		return string.format("%.2fM", number / 1000000)
	elseif number >= 1000 then
		return string.format("%.1fk", number / 1000)
	else
		return tostring(number)
	end
end

		corner.CornerRadius = UDim.new(1, 0)
		corner.Parent = particle

		particle.Parent = gui

		local angle = math.rad(math.random(0, 360))
		local distance = math.random(100, 500)

		local xOffset = math.cos(angle) * distance
		local yOffset = math.sin(angle) * distance

		local tweenInfo = TweenInfo.new(math.random(15, 30)/10, Enum.EasingStyle.Quart, Enum.EasingDirection.Out)
		local tween = TweenService:Create(particle, tweenInfo, {
			Position = UDim2.new(0.5, xOffset, 0.5, yOffset + 350),
			Rotation = math.random(-360, 360),
			BackgroundTransparency = 1
		})
		tween:Play()
		Debris:AddItem(particle, 3) 
	end
end

local function ShowPreview(caseName)
	local case = CaseData.Cases[caseName]
	if not case then return end

	selectedCase = caseName
	currentCasePrice = case.Price

	previewTitle.Text = string.upper(caseName)

	local formattedPrice = tostring(case.Price):reverse():gsub("%d%d%d", "%1,"):reverse():gsub("^,", "")
	previewSpinBtn.Text = "SPIN - $" .. formattedPrice

	for _, child in pairs(previewGrid:GetChildren()) do
		if child:IsA("Frame") and child.Name ~= "ItemTemplate" then
			child:Destroy()
		end
	end

	local totalWeight = 0
	for _, w in pairs(case.Weights) do
		totalWeight += w
	end

	local previewItems = {}
	for rarity, items in pairs(case.Pool or {}) do
		local rWeight = case.Weights[rarity] or 0
		local chanceToShow = (rWeight / totalWeight) * 100

		-- 🔥 ROBLOX TOS FIX: ROZDELENIE ŠANCE MEDZI JEDNOTLIVÉ PREDMETY 🔥
		local pocetItemovVRarite = #items
		local skutocnaSanca = chanceToShow / pocetItemovVRarite

		for _, itemName in ipairs(items) do
			local data = CaseData.ItemDatabase[itemName]
			if data then
				table.insert(previewItems, {
					Name = itemName,
					Rarity = rarity,
					Image = data.Image,
					Chance = skutocnaSanca,
					Priority = rarityPriority[rarity] or 99
				})
			end
		end
	end

	table.sort(previewItems, function(a, b)
		if a.Priority ~= b.Priority then
			return a.Priority < b.Priority
		else
			return a.Name < b.Name
		end
	end)

	for _, item in ipairs(previewItems) do
		local slot = previewTemp:Clone()
		slot.Name = item.Name
		slot.Visible = true
		slot.Parent = previewGrid

		if slot:FindFirstChild("ItemName") then
			slot.ItemName.Text = item.Name
		end
		if slot:FindFirstChild("ItemIcon") then
			slot.ItemIcon.Image = item.Image or "rbxassetid://134566735"
		end

		local rColor = CaseData.RarityStats[item.Rarity] and CaseData.RarityStats[item.Rarity].Color or Color3.new(1,1,1)
		if slot:FindFirstChild("UIStroke") then
			slot.UIStroke.Color = rColor
		end
		if slot:FindFirstChild("ItemName") then
			slot.ItemName.TextColor3 = rColor
		end

		local chanceText = ""
		if item.Chance < 0.01 then
			chanceText = "<0.01%"
		else
			chanceText = string.format("%.2f%%", item.Chance)
		end
		if slot:FindFirstChild("Chance") then
			slot.Chance.Text = chanceText
		end
	end

	if previewSpinConnection then
		previewSpinConnection:Disconnect()
	end
	previewSpinConnection = previewSpinBtn.MouseButton1Click:Connect(function()
		previewWindow.Visible = false
		AttemptSpin(selectedCase, currentCasePrice)
	end)

	previewWindow.Visible = true
end

local function GetRandomVisualItem(caseName)
	local case = CaseData.Cases[caseName]
	if not case or not case.Pool then 
		return { Name = "???", Image = "rbxassetid://134566735", Rarity = {Color = Color3.fromRGB(100,100,100)} } 
	end
	local rarities = {}
	for r, _ in pairs(case.Pool) do table.insert(rarities, r) end
	if #rarities == 0 then return { Name = "???", Image = "rbxassetid://134566735", Rarity = {Color = Color3.fromRGB(100,100,100)} } end

	local r = rarities[math.random(1, #rarities)]
	local items = case.Pool[r]
	local iName = items[math.random(1, #items)]
	local dbData = CaseData.ItemDatabase[iName]

	return {
		Name = iName,
		Image = dbData and dbData.Image or "rbxassetid://134566735",
		Rarity = {Name = r, Color = CaseData.RarityStats[r] and CaseData.RarityStats[r].Color or Color3.fromRGB(150,150,150)}
	}
end

			autoRollToggle.Text = "AUTO-ROLL: OFF"
			autoRollToggle.TextColor3 = Color3.fromRGB(255, 80, 80)
			if autoRollToggle:FindFirstChildOfClass("UIStroke") then
				autoRollToggle.UIStroke.Color = Color3.fromRGB(255, 80, 80)
			end
		end

		openBtn.Visible = false
	end

			local dots = 0

			novyText.Name = "ItemName"
			novyText.AnchorPoint = Vector2.new(0.5, 0.5)
			novyText.Position = UDim2.new(0.5, 0, 0.82, 0)
			novyText.Size = UDim2.new(0.95, 0, 0.30, 0) 
			novyText.BackgroundTransparency = 1
			novyText.Text = itemToShow.Name or "???"
			novyText.TextScaled = true
			novyText.TextWrapped = true
			novyText.Font = Enum.Font.GothamBlack
			novyText.TextColor3 = Color3.fromRGB(0, 0, 0) 
			novyText.ZIndex = 5 
			novyText.Parent = card

			-- 🛠️ ZABRÁNÍ PŘETEČENÍ TEXTU
			local constraint = Instance.new("UITextSizeConstraint")
			constraint.MaxTextSize = 20
			constraint.MinTextSize = 8
			constraint.Parent = novyText
		end)

		if not success then
			warn("Chyba pri generování karty: " .. tostring(err))
		end

		-- 🛠️ ZÁCHRANNÁ BRZDA PROTI ZÁSEKU: Roblox si po každých 15 kartách na mikrosekundu oddechne
		if i % 15 == 0 then
			RunService.RenderStepped:Wait()
		end
	end

	spinner.CanvasSize = UDim2.new(0, TOTAL_ITEMS * FULL_CARD_WIDTH, 0, 0)

	local winnerPos = (WINNER_INDEX - 1) * FULL_CARD_WIDTH
	local centerOffset = spinner.AbsoluteSize.X / 2
	local cardCenter = CARD_WIDTH / 2
	local targetX = winnerPos - centerOffset + cardCenter + math.random(-40, 40)

	local tweenInfo = TweenInfo.new(SPIN_TIME, Enum.EasingStyle.Quart, Enum.EasingDirection.Out)
	local tween = TweenService:Create(spinner, tweenInfo, {CanvasPosition = Vector2.new(targetX, 0)})

	local sounds = workspace:FindFirstChild("GameSounds")
	local connection
	if sounds and sounds:FindFirstChild("SpinLoop") then
		local sound = sounds.SpinLoop
		local lastCardIndex = 0
		local lastSoundTime = 0 

		local playCaseSounds = player:GetAttribute("Pref_MutationVol")
		if playCaseSounds == nil then
			playCaseSounds = true
		end

		connection = RunService.RenderStepped:Connect(function()
			local currentX = spinner.CanvasPosition.X
			local currentCardIndex = math.floor(currentX / FULL_CARD_WIDTH)
			if currentCardIndex > lastCardIndex then
				local now = tick()
				if (now - lastSoundTime) > 0.04 then
					if playCaseSounds then
						sound:Play()
					end
					lastSoundTime = now
				end
				lastCardIndex = currentCardIndex
			end
		end)
	end

	tween:Play()
	tween.Completed:Wait() 

	if connection then
		connection:Disconnect()
	end 
	ShowReward(result)
end

				sounds.WinLegendary:Play()
